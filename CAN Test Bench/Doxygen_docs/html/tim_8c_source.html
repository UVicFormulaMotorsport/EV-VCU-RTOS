235void handle_wheel_interrupt(uint32_t wheel_index)
236{
237 uint32_t now = __HAL_TIM_GET_COUNTER(&htim5);
238 uint32_t last = last_timestamp[wheel_index];
239
240 if (now >= last) {
241 period[wheel_index] = now - last;
242 } else {
243 period[wheel_index] = (0xFFFFFFFF - last) + now; // Handle timer overflow
244 }
245 last_timestamp[wheel_index] = now;
246}
