46int32_t sprint_fixed_d(char* buf, const char* label, int32_t value, int decimals, const char* unit)
47{
48 int32_t nchars_written = 0;
49 int32_t tmp = 0;
50 int32_t scale = 1;
51 for(int i = 0; i < decimals; i++) scale *= 10;
52
53 int32_t whole = value / scale;
54 int32_t frac = value % scale;
55 if(frac < 0) frac = -frac;
56
57 tmp = sprintf(buf,"%s: %ld", label, (long)whole);
58 if(tmp < 0){
59 return -1;
60 }else{
61 nchars_written += tmp;
62 buf += tmp;
63 }
64 if(decimals > 0){
65 tmp = sprintf(buf,".%0*ld", decimals, (long)frac);
66 if(tmp < 0){
67 return -1;
68 }
69 nchars_written += tmp;
70 buf += tmp;
71 }
72 if(unit){
73 tmp = sprintf(buf," %s", unit);
74 if(tmp < 0){
75 return -1;
76 }
77 nchars_written += tmp;
78 buf += tmp;
79 }
80 tmp = sprintf(buf,"\n");
81 nchars_written += tmp;
82 buf += tmp;
83
84
85 return nchars_written;
86}

216void uvGetSDCStat(){
217 //IMD - 0 on pin is bad
218 if(!HAL_GPIO_ReadPin(GPIOD, GPIO_PIN_1)){
219 g_imd_fault = 1;
220 }else{
221 g_imd_fault = 0;
222 }
223
224 //BMS - Open drain -> 1 bad
225 if(HAL_GPIO_ReadPin(GPIOD, GPIO_PIN_2)){
226 g_bms_fault = 1;
227 }else{
228 g_bms_fault = 0;
229 }
230
231 //BSPD - Open drain -> 1 bad
232 if(HAL_GPIO_ReadPin(GPIOD, GPIO_PIN_3)){
233 g_bspd_fault = 1;
234 }else{
235 g_bspd_fault = 0;
236 }
237
238 g_SDCR_tripped = g_bms_fault || g_bspd_fault || g_imd_fault;
239}
