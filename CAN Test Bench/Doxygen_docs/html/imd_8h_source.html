17enum imd_status_bits{
18 // I have just left this as plain binary. Could also be declared by bitshifting but I think this is easier
19 // 12345678
20 Isolation_status_bit0 = 0b00000001, // IS0
21 Isolation_status_bit1 = 0b00000010, // IS1
22 Low_Battery_Voltage = 0b00000100, // LV 15V threshold (1 if below 15V)
23 High_Battery_Voltage = 0b00001000, // HV (1 if higher than Max_battery_working_voltage)
24 Exc_off = 0b00010000, // EO (Excitation pulse 0 for operating)
25 High_Uncertainty = 0b00100000, // HU (1 for greater than 5%)
26 Touch_energy_fault = 0b01000000, // EF (1 for energy exceeds 0.2J)
27 Hardware_Error = 0b10000000, // HE (1 for error)
28};

38// The MCU needs to send a message to the IMD requesting info
39// These are requests that will return status bits (defined above) & the value requested
40enum imd_status_requests{
41 // The electrical isolation is in bytes 2 & 3
42 uv_request_mux_isolation_state = 0xE0,
43
44 // Rp is the resistance from the positive of the battery to chassis
45 // Rn is the resistance from the negative of the battery to chassis
46 uv_request_mux_isolation_resistances = 0xE1,
47
48 // Cp is the capacitance from the positive of the battery to chassis
49 // Cn is the capacitance from the negative of the battery to chassis
50 // Bytes 2 & 3 are Cp, bytes 5 & 6 are Cn, in nF
51 uv_request_mux_isolation_capacitances = 0xE2,
52
53 // High voltage battery voltages to chassis
54 uv_request_mux_voltages_Vp_and_Vn = 0xE3,
55
56 // GLV battery voltage
57 uv_request_mux_battery_voltage = 0xE4,
58
59 // Error flags are a series of bits described below
60 uv_request_mux_Error_flags = 0xE5,
61
62 // The IMD monitors charge stored in the system
63 uv_request_mux_safety_touch_energy = 0xE6,
64
65 // Also monitors if it is safe to touch
66 uv_request_mux_safety_touch_current = 0xE7,
67
68 // This is just a parameter we can set
69 uv_request_mux_Max_battery_working_voltage = 0xF0,
70
71 // We can read the temperature of the board
