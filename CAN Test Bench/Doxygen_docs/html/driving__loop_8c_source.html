450/*
451 * Adaptive Torque Map (Drive Only)
452 *
453 * 0) Normalize throttle (0–100% → 0–1) and remove deadband.
454 *
455 * 1a) Speed-dependent pedal shift ("moving map"):
456 * - Compute a zero-torque pedal threshold based on RPM.
457 * - Low RPM → larger zero-torque region.
458 * - High RPM → smaller zero-torque region.
459 * - If pedal <= threshold → 0 Nm.
460 * - Otherwise shift + rescale remaining pedal travel to 0–1.
461 *
462 * 1b) Apply pedal shaping (Linear / Exp / Cubic)
463 * - Shapes driver feel only.
464 * - Does NOT enforce torque limits.
465 *
466 * 3) Optional high-RPM softening:
467 * - Above soften_rpm, torque is gradually reduced toward max_rpm.
468 *
469 * 4) Final torque request:
470 * T_req = T_allow * shaped_pedal
471 *
472 * Result:
473 * Speed-aware pedal behavior with mode-dependent feel,
474 * while torque ceilings are enforced upstream.
475 */
476static float mapAdaptiveFromMode(float throttle_percent, float T_max,
477 const driving_loop_args* dl,
478 const drivingMode* dm)
479{
480 /*
481 * STEP 0: Normalize throttle input to 0–1 with deadband
482 * apps = throttle as fraction (0–1)
483 * dead = deadband fraction (0–1)
484 * After deadband removal:
485 * x = 0 at bottom of pedal
486 * x = 1 at full pedal
487 */
488 float apps = dl_clampf(throttle_percent / 100.0f, 0.0f, 1.0f);
489 const float dead = dl_clampf(dl->throttle_deadband_percent / 100.0f, 0.0f, 0.9f);
490 float x = (apps - dead) / (1.0f - dead);
491 x = dl_clampf(x, 0.0f, 1.0f);
492 // If below deadband, request zero torque immediately
493 if (x <= 0.0f) return 0.0f;
494
495 /* =======================================================================
496 * STEP 1A: Speed-Dependent Pedal Map Shift ("Moving Map")
497 *
498 * The pedal has a coasting region (zero torque zone) that depends on speed.
499 *
500 * - At low RPM → larger zero-torque region (more pedal travel = coast)
501 * - At higher RPM → smaller zero-torque region (torque engages sooner)
502 *
503 * We:
504 * 1) Compute the speed-dependent zero-torque threshold
505 * 2) If pedal is inside that region → return 0 torque
506 * 3) Otherwise shift and rescale pedal so remaining travel maps 0–1
507 * ======================================================================= */
508
509 extern int16_t mc_speed_rpm;
510
511 /* Use absolute RPM (forward/reverse behave the same for shaping) */
512 float vehicle_rpm = (float)mc_speed_rpm;
513 if (vehicle_rpm < 0.0f)
514 vehicle_rpm = -vehicle_rpm;
515
516
517 /* === Per-mode tuning parameters === */
518
519 // [Ratio 0.0 to 1.0] The "Deadzone" size at 0 RPM (e.g., 0.05 = 5%).
520 // This provides a safety buffer so the car doesn't "creep" at a standstill.
521 float zero_torque_at_low_speed = dm->adaptive_settings.coast_p_low;
522
523 // [Ratio 0.0 to 1.0] The "Deadzone" size at high speed (e.g., 0.02 = 2%).
524 // Shrinking the deadzone here makes the car feel more responsive once moving.
525 float zero_torque_at_high_speed = dm->adaptive_settings.coast_p_high;
526
527 // [RPM] The motor speed where the map starts moving (usually 0 RPM).
528 float low_speed_rpm =(float)dm->adaptive_settings.coast_rpm_start;
529
530 // [RPM] The motor speed where the map reaches its final "high speed" shape (e.g., 1200 RPM).
531 float high_speed_rpm =(float)dm->adaptive_settings.coast_rpm_end;
532
533 /* Keep values safe (Clamps to 0.0-0.9 range to ensure the pedal always works) */
534 zero_torque_at_low_speed = dl_clampf(zero_torque_at_low_speed, 0.0f, 0.9f);
535 zero_torque_at_high_speed = dl_clampf(zero_torque_at_high_speed, 0.0f, 0.9f);
536
537 /* === Compute speed-dependent zero-torque threshold === */
538
539 /* THE ZERO TORQUE THRESHOLD:
540 * This is the "Live Edge." It is the specific point on the pedal travel where
541 * the motor transitions from doing nothing (coasting) to applying torque.
542 */
543 float zero_torque_threshold = zero_torque_at_high_speed;
544
545 /* CHECK: (high_speed_rpm > low_speed_rpm)
546 * This ensures the tuning range is valid to avoid dividing by zero in the math below.
547 */
548 if (high_speed_rpm > low_speed_rpm){
549
550 /* THE SPEED FRACTION (The "Where am I?" Variable):
551 * Tells us how far we are in the transition from 0 to 1200 RPM.
552 * 0.0 = At/below Start RPM. 0.5 = Halfway (600 RPM). 1.0 = At/above End RPM.
553 */
554 float speed_fraction =(vehicle_rpm - low_speed_rpm) /(high_speed_rpm - low_speed_rpm);
555 speed_fraction = dl_clampf(speed_fraction, 0.0f, 1.0f);
556
557 /* INTERPOLATE:
558 * Slides the threshold between Low (5%) and High (2%) based on the speed_fraction.
559 * As you speed up, the threshold (deadzone) gets smaller.
560 */
561 zero_torque_threshold =(1.0f - speed_fraction) * zero_torque_at_low_speed + speed_fraction * zero_torque_at_high_speed;
562 }
563
564
565 /* === Apply coasting region === */
566
567 /* THE COAST GATE:
568 * If current pedal position (x) is below the threshold we just calculated, force 0 Nm.
569 * This is the "Electronic Coast" zone where the car rolls freely.
570 */
571 if (x <= zero_torque_threshold)
572 return 0.0f; // [Nm] Return zero torque
573
574
575 /* THE SHIFT AND RESCALE:
576 * Stretches the remaining pedal travel (from threshold to 100%) back to a 0.0-1.0 range.
577 * This prevents a "jump" in torque when you cross the threshold.
578 * Formula: (Foot_Position - Deadzone) / (Available_Pedal_Range)
579 */
580 float shifted_pedal = (x - zero_torque_threshold) / (1.0f - zero_torque_threshold);
581
582
583 shifted_pedal = dl_clampf(shifted_pedal, 0.0f, 1.0f);
584
585 /* Use shifted_pedal (x) for the final torque curves (Linear, Exponential, etc.) */
586 x = shifted_pedal;
587
588 /*

590 * curve_type selects how throttle maps to torque fraction:
