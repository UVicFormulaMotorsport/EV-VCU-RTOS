591uv_status uvIsPTRValid(void* ptr){
592 if(ptr == NULL){
593 return UV_WARNING;
594 }
595 uint32_t pval = (uint32_t)ptr;
596
597 //bool is_valid = false;
598
599 if(pval < 0x000FFFFF){ //Aliased to FLASH, systmem or SRAM
600 return UV_OK;
601 }
602
603 if((pval > 0x08000000)&& (pval < 0x080FFFFF)){ //Flash be like
604 return UV_OK;
605 }
606
607 if((pval > 0x10000000)&&(pval < 0x1000FFFF)){ //CCM Data RAM
608 return UV_OK;
609 }
610
611 if((pval > 0x1FFF0000)&&(pval < 0x1FFF7A0F)){ //System memory + OTP
612 return UV_OK;
613 }
614
615 if((pval > 0x1FFFC000)&&(pval < 0x1FFFC007)){ //option bytes (should these be user accessable under any circumstances?)
616 return UV_WARNING;
617 }
618
619 if((pval > 0x20000000)&&(pval < 0x2001FFFF)){ //SRAM :)
620 return UV_OK;
621 }
622
623 if((pval > 0x40000000)&&(pval < 0x40007FFF)){ //APB1
624 return UV_OK;
625 }
626
627 if((pval > 0x40010000)&&(pval < 0x400157FF)){ //APB2
628 return UV_OK;
629 }
630
631 if((pval > 0x40020000)&&(pval < 0x4007FFFF)){ //AHB1
632 return UV_OK;
633 }
634
635 if((pval > 0x50000000)&&(pval < 0x50060BFF)){//AHB2
636 return UV_OK;
637 }
638
639 if((pval > 0x60000000)&&(pval < 0xA0000FFF)){//AHB3
640 return UV_OK;
641 }
642
643 if((pval > 0xE0000000)&&(pval < 0xE00FFFFF)){//
644 return UV_OK;
645 }
646
647 return UV_ERROR;
648}
