1/*
2 * tms.c
3 *
4 * VCU-side Thermal Management System (TMS) tap. See tms.h for the full rundown.
5 *
6 * In short: the TMS broadcasts battery-pack temperatures as a J1939 thermistor
7 * frame (0x1839F380) aimed at the BMS, but mirrored onto both buses. The BMS
8 * handles its own copy; we just listen in on the dash bus, sanity-check the
9 * frame, and drop the numbers into g_tms_state so the dash/DAQ can display them.
10 * Nothing here is safety-critical -- it is display/telemetry only.
11 */

30void TMS_CANRxHandler(uv_CAN_msg* msg){
31 if(msg == NULL){
32 return;
33 }
34
35 /* Checksum: byte7 = low 8 bits of (sum of bytes 0..6 + 0x39 + DLC(8)).
36 * This mirrors can_tms_send_module_data() on the TMS side exactly. */
37 uint16_t sum = (uint16_t)msg->data[0] + msg->data[1] + msg->data[2]
38 + msg->data[3] + msg->data[4] + msg->data[5] + msg->data[6]
39 + 0x39u + 0x08u;
40
41 if((uint8_t)(sum & 0xFFu) != msg->data[7]){
42 g_tms_state.msg_corrupt = 1; // keep the last good values, drop this frame
43 return;
44 }
45
46 g_tms_state.msg_corrupt = 0;
47 g_tms_state.low_c = (int8_t)msg->data[1];
48 g_tms_state.high_c = (int8_t)msg->data[2];
49 g_tms_state.avg_c = (int8_t)msg->data[3];
50 g_tms_state.therm_count = msg->data[4];
51 g_tms_state.high_id = msg->data[5];
52 g_tms_state.low_id = msg->data[6];
53
54 /* Let XDEVMON know the TMS is alive (freshness tracking only). */
55 externalDeviceRxHandler(TMS);
56}
