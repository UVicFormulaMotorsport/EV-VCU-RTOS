1/*
2 * tms.h
3 *
4 * VCU-side handler for the Thermal Management System (TMS).
5 *
6 * The TMS impersonates an Orion external thermistor expansion module and talks
7 * to the BMS over J1939 entirely on its own -- the BMS ingests those frames
8 * directly, so the VCU is NOT part of that thermal safety loop and does not need
9 * to handle the TMS->BMS path. The TMS happens to publish the same frames on
10 * both CAN buses, so VCU-side we simply listen in on the dash bus (CAN2) and
11 * stash the pack temperatures in g_tms_state. This is a display/telemetry tap
12 * only -- it exists so the temps can be shown on the dash and logged via DAQ.
13 *
14 * Frame we care about (see EV26-TMS-NEW/TMS_MAIN/Core/Src/can_tms.c):
15 * 0x1839F380 Thermistor module -> BMS @ ~100 ms (extended ID, DLC=8)
16 * byte0 = module# byte4 = therm_count (always 6)
17 * byte1 = low_c (int8) byte5 = high_id (pack 0..5)
18 * byte2 = high_c (int8) byte6 = low_id (pack 0..5)
19 * byte3 = avg_c (int8) byte7 = checksum
20 *
21 * Temperatures are int8 degrees C (1 C / LSB, signed). Each pack value is itself
22 * a hottest-cell reading, so low_c/avg_c run warm by design.
23 */

42/* Runtime state (telemetry). Mirrors can_tms_module_data_t on the TMS side. */
43typedef struct {
44 int8_t low_c; // lowest pack reading, signed degrees C
45 int8_t high_c; // highest pack reading, signed degrees C
46 int8_t avg_c; // mean of the pack readings, signed degrees C
47 uint8_t therm_count; // number of packs reported (expected 6)
48 uint8_t high_id; // pack index (0..5) reporting high_c
49 uint8_t low_id; // pack index (0..5) reporting low_c
